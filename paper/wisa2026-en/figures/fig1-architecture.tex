% Coordinates fit LNCS textwidth (~12cm) directly without \resizebox.
% Z3 admission is the central gate; production path (top row) flows
% rules -> Z3 -> compiled -> JIT -> diff tests; audit path (bottom row)
% flows enum -> Z3 -> audit (dashed; not deployed to JIT).
\begin{figure}[t]
\centering
\begin{tikzpicture}[
  every node/.style={font=\scriptsize},
  box/.style={draw, rounded corners=2pt, inner sep=3pt, minimum height=8mm, minimum width=15mm, align=center, fill=gray!5},
  z3box/.style={draw, rounded corners=2pt, inner sep=3pt, minimum height=8mm, minimum width=14mm, align=center, fill=blue!8},
  audit/.style={draw, rounded corners=2pt, inner sep=3pt, minimum height=8mm, minimum width=15mm, align=center, fill=gray!5, dashed},
  arrow/.style={-{Latex[length=1.6mm]}, semithick},
  arrowdash/.style={-{Latex[length=1.6mm]}, semithick, dashed, gray!70},
  region/.style={font=\footnotesize\bfseries},
  small/.style={font=\fontsize{6}{7}\selectfont},
]

% --- Production-deployment path (top row) ---
\node[box]   (rules)    at (0,    0.9)  {Hand-curated rules\\{\small[70 dMIR + 13 x86]}};
\node[z3box] (z3)       at (3.0,  0.0)  {Z3 admission\\{\small(gate)}};
\node[box]   (compiled) at (5.6,  0.9)  {Compiled\\rule set};
\node[box]   (jit)      at (8.2,  0.9)  {JIT runtime\\{\small(dMIR+x86)}};
\node[box]   (diff)     at (10.8, 0.9)  {5{,}884 diff.\\Cancun tests};

% --- Audit / measurement path (bottom row) ---
\node[box]   (enum)  at (0,    -0.9) {Canonical-LHS enum.\\{\small(853 / 1{,}966)}};
\node[audit] (audit) at (5.6,  -0.9) {1{,}966 admitted\\{\small(audit; not deployed)}};

% Region labels (above their cluster, no separator lines)
\node[region] at (1.5,  1.7) {Offline};
\node[region] at (6.9,  1.7) {JIT runtime};
\node[region] at (10.8, 1.7) {Correctness gate};

% Edges -- production deployment (corner-to-corner direct lines)
\draw[arrow] (rules.east)    -- (z3.north west);
\draw[arrow] (z3.north east) -- (compiled.west);
\draw[arrow] (compiled.east) -- (jit.west);
\draw[arrow] (jit.east)      -- (diff.west);

% Edges -- audit channel (dashed to convey "not deployed")
\draw[arrowdash] (enum.east)    -- (z3.south west);

\end{tikzpicture}
\caption{Two-tier system overview. Hand-curated production rules
(\textbf{70} dMIR + \textbf{13} x86) pass an offline Z3 admission
gate (admit rates: 98.7\% algebraic, 21.8\% carry-template) and
become the compiled rule set deployed by the JIT (dMIR matcher
followed by x86 CgIR cleanup); correctness is gated by 5{,}884
differential Cancun tests. In parallel, canonical-LHS enumeration
submits 853 capacity-run candidates and produces \textbf{1{,}966}
admitted rewrites whose canonical left-hand sides cover all 70
production rules; the audit channel (dashed) is a measurement
output and is \emph{not} deployed to the JIT.}
\label{fig:architecture}
\end{figure}
