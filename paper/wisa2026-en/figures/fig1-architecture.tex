% figures/fig1-architecture.tex — two-tier system overview
% Layout fits LNCS textwidth (~11.5cm) without \resizebox.
% The Offline region (left) holds rules + Z3 + enum + audit; the audit
% box sits directly below Z3 in the same column to keep the
% measurement channel visually inside Offline rather than under JIT
% runtime. JIT runtime region holds compiled + jit; Correctness gate
% holds diff. Audit-channel edges drawn dashed black to distinguish
% from solid production-path arrows while remaining visible.
\begin{figure}[t]
\centering
\begin{tikzpicture}[
  every node/.style={font=\scriptsize},
  box/.style={draw, rounded corners=2pt, inner sep=3pt, minimum height=8mm, minimum width=15mm, align=center, fill=gray!5},
  z3box/.style={draw, rounded corners=2pt, inner sep=3pt, minimum height=8mm, minimum width=15mm, align=center, fill=blue!8},
  audit/.style={draw, rounded corners=2pt, inner sep=3pt, minimum height=8mm, minimum width=15mm, align=center, fill=gray!5, dashed},
  arrow/.style={-{Latex[length=1.6mm]}, semithick},
  arrowdash/.style={-{Latex[length=1.6mm]}, semithick, dashed},
  region/.style={font=\footnotesize\bfseries},
]

% Production-deployment path
\node[box]   (rules)    at (0,    0.9)  {Hand-curated rules\\{\small[70 dMIR + 13 x86]}};
\node[z3box] (z3)       at (2.5,  0.0)  {Z3 admission};
\node[box]   (compiled) at (5.0,  0.0)  {Compiled\\rule set};
\node[box]   (jit)      at (7.5,  0.0)  {JIT runtime\\{\small(dMIR+x86)}};
\node[box]   (diff)     at (10.0, 0.0)  {5{,}884 diff.\\Cancun tests};

% Audit / measurement path (under Offline region)
\node[box, dashed] (enum)  at (0,    -1.1) {Canonical-LHS enum.\\{\small(853 / 1{,}966)}};
\node[audit]       (audit) at (2.5,  -1.1) {1{,}966 admitted\\{\small(audit; not deployed)}};

% Region labels
\node[region] at (1.25, 1.7) {Offline};
\node[region] at (6.25, 1.7) {JIT runtime};
\node[region] at (10.0, 1.7) {Correctness gate};

% Production edges
\draw[arrow] (rules.south east) -- (z3.north west);
\draw[arrow] (z3.east)          -- (compiled.west);
\draw[arrow] (compiled.east)    -- (jit.west);
\draw[arrow] (jit.east)         -- (diff.west);

% Audit edges (dashed black, clearly visible)
\draw[arrowdash] (enum.east)    -- (z3.south west);
\draw[arrowdash] (z3.south)     -- (audit.north);

\end{tikzpicture}
\caption{Two-tier system overview. Hand-curated rules and a separate
canonical-LHS enumeration both pass through the Z3 admission gate;
the production set is compiled into the JIT, and correctness is
gated by 5{,}884 differential Cancun tests. The enumeration channel
(dashed) produces 1{,}966 audit rules used only for coverage
measurement, not deployment.}
\label{fig:architecture}
\end{figure}
